 *

 *
 * ?case=s1       the #40 repro: a resume the disk killed        (leg A capture)
 * ?case=s2       a resume GitHub refused the create for         (leg A capture)
 * ?case=s3       a lookup that could not answer at all          (leg A capture)
 * ?case=s4       pending, then [Check again] resolves it        (leg A capture)
 * ?case=before   the row #29 shipped, for comparison            (#29's log)
 * &delay=ms      how long the pass takes before it resolves
 *
 * Delete after the run.
 */

const wait = (ms: number) => new Promise((r) => setTimeout(r, ms));

// ── The recovery script ─────────────────────────────────────────────────────
//
// One entry per report a previous run left behind. `passes[n]` is what the
// n-th `recover()` for that report says — pass 0 is startup, pass 1 is what
// [Check again] gets. The last pass repeats if asked again.
//
// For #40 every pass is READ OFF DISK: it is exactly what leg A watched the
// shipped `createFiling().recover()` emit against real `gh` and a real disk.

interface Plan {
  id: string;
  passes: RecoveryEvent[][];
}

/** The row #29 shipped, transcribed from docs/qa/29/leg-a-real-gh.log (S4). */
const BEFORE: RecoveryEvent[] = [
  { state: 'checking', filingId: 'fil_qa29_s4_before' },
  { state: 'pending', filingId: 'fil_qa29_s4_before', message: 'Could not file this report.' },
];

/**
 * A capture is either one pass or a pass per look. `s4-events.json` is the
 * second shape: what startup said, then what [Check again] got.
 */
const loadPasses = async (): Promise<RecoveryEvent[][]> => {
  if (kase === 'before') return [BEFORE];
  const res = await fetch(`/docs/qa/40/${kase}-events.json`);
  if (!res.ok) throw new Error(`no capture for ?case=${kase} (${res.status})`);
  const body = (await res.json()) as RecoveryEvent[] | RecoveryEvent[][];
  return Array.isArray(body[0]) ? (body as RecoveryEvent[][]) : [body as RecoveryEvent[]];
};

const captured = await loadPasses();
const plans: Plan[] = [{ id: captured[0][0]?.filingId ?? 'fil_unknown', passes: captured }];
const passCount = new Map<string, number>();

// ── Harness recorder ────────────────────────────────────────────────────────

const strip = document.getElementById('harness')!;
const lines: Array<{ text: string; n: number }> = [];
/** Consecutive repeats collapse: auto-save fires per keystroke and 40 identical
 *  rows say nothing 1 row and a count does not. */
const record = (line: string): void => {
  const last = lines[lines.length - 1];
  if (last?.text === line) last.n += 1;
  else lines.push({ text: line, n: 1 });
  strip.innerHTML =
    '<b>HARNESS LOG — what the app asked the OS/disk to do (not app UI)</b><br>' +
    lines.map((l) => `<div>${l.text}${l.n > 1 ? ` [x${l.n}]` : ''}</div>`).join('');
};

/**
 * Mirrors the real `recover()`'s in-flight guard: a Filing a pass is already
 * working on is skipped, so a second concurrent call can never double-create.
 * Without it, React StrictMode's dev double-mount would eat two scripted passes
 * at startup and no [Check again] could ever be observed.
 */
const inFlight = new Set<string>();

async function* recover(only?: string): AsyncGenerator<RecoveryEvent> {
  record(`recover(${only ?? 'all'}) called`);
  for (const plan of plans) {
    if (only !== undefined && plan.id !== only) continue;
    if (inFlight.has(plan.id)) {
      record(`  ↳ ${plan.id} skipped — a look is already in flight`);
      continue;
    }
    inFlight.add(plan.id);
    const n = passCount.get(plan.id) ?? 0;
    passCount.set(plan.id, n + 1);
    const pass = plan.passes[Math.min(n, plan.passes.length - 1)];
    for (const [i, event] of pass.entries()) {
      if (i > 0) await wait(delay);
      yield event;
    }
    // A pass whose script stops at `checking` is a lookup that never answers.
    if (pass[pass.length - 1]?.state === 'checking') await new Promise<void>(() => {});
    inFlight.delete(plan.id);
  }
}

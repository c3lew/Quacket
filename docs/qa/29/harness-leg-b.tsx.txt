/**
 * THROWAWAY QA harness for #29, leg B: the real palette in a real browser.
 *
 * Mounts the shipped <App> with a hand-written UiServices. Nothing about the
 * Tauri shell is involved — App imports nothing host-specific — so what is on
 * screen is the real component tree reacting to a real `recover()` stream and
 * the real reducer deciding what to say and when to notify.
 *
 * The ONE thing added to the page is a labelled harness strip at the bottom
 * recording what the app asked the OS to do (notifications, opened URLs, draft
 * writes) — a browser has no tray, so the effect is shown instead of fired.
 *
 * ?case=pending   a previous report GitHub could not be asked about
 * ?case=hang      the lookup never comes back
 * ?case=failed    a previous report provably never reached GitHub
 * ?case=filed     it turns out to have been filed
 * ?case=recheck   pending first, filed on the second look
 * ?case=two       two unresolved reports, so [Check again] scope can be seen
 * &delay=ms       how long each pass takes before it resolves
 *
 * Delete after the run.
 */

const params = new URLSearchParams(location.search);

const delay = Number(params.get('delay') ?? '1200');

const wait = (ms: number) => new Promise((r) => setTimeout(r, ms));

const receipt = (n: number, id: string, kind: 'new-issue' | 'comment' = 'new-issue'): FilingReceipt => ({
  url: `https://github.com/${REPO}/issues/${n}`,
  issueNumber: n,

// ── The recovery script ─────────────────────────────────────────────────────
//
// One entry per report a previous run left behind. `passes[n]` is what the
// n-th `recover()` for that report says — pass 0 is startup, pass 1 is what
// [Check again] gets. The last pass repeats if asked again.

interface Plan {
  id: string;
  passes: RecoveryEvent[][];
}

const OFFLINE = 'Could not check GitHub for this report.';

const checking = (id: string): RecoveryEvent => ({ state: 'checking', filingId: id });
const pending = (id: string): RecoveryEvent => ({ state: 'pending', filingId: id, message: OFFLINE });
const filed = (id: string, n: number, kind: 'new-issue' | 'comment' = 'new-issue'): RecoveryEvent => ({
  state: 'filed',
  filingId: id,
  receipt: receipt(n, id, kind),
});
const failedEv = (id: string): RecoveryEvent => ({
  state: 'failed',
  filingId: id,
  kind: 'upload_failed',
  message: 'Could not upload an image to this repo.',
});

const A = 'fil_qa29_a';
const B = 'fil_qa29_b';

const PLANS: Record<string, Plan[]> = {
  pending: [{ id: A, passes: [[checking(A), pending(A)]] }],
  /** The exact message a locally-failed create emits (leg A S4), composed. */
  contradiction: [
    {
      id: A,
      passes: [
        [checking(A), { state: 'pending', filingId: A, message: 'Could not file this report.' }],
      ],
    },
  ],
  hang: [{ id: A, passes: [[checking(A)]] }],
  failed: [{ id: A, passes: [[checking(A), failedEv(A)]] }],
  filed: [{ id: A, passes: [[checking(A), filed(A, 57)]] }],
  comment: [{ id: A, passes: [[checking(A), filed(A, 37, 'comment')]] }],
  recheck: [{ id: A, passes: [[checking(A), pending(A)], [checking(A), filed(A, 58)]] }],
  two: [
    { id: A, passes: [[checking(A), pending(A)], [checking(A), filed(A, 59)]] },
    { id: B, passes: [[checking(B), pending(B)], [checking(B), filed(B, 60)]] },
  ],
};

const plans = PLANS[kase] ?? PLANS.pending;
const passCount = new Map<string, number>();

// ── Harness recorder ────────────────────────────────────────────────────────

const strip = document.getElementById('harness')!;
const lines: Array<{ text: string; n: number }> = [];
/** Consecutive repeats collapse: auto-save fires per keystroke and 40 identical
 *  rows say nothing 1 row and a count does not. */
const record = (line: string): void => {
  const last = lines[lines.length - 1];
  if (last?.text === line) last.n += 1;
  else lines.push({ text: line, n: 1 });
  strip.innerHTML =
    '<b>HARNESS LOG — what the app asked the OS/disk to do (not app UI)</b><br>' +
    lines.map((l) => `<div>${l.text}${l.n > 1 ? ` [x${l.n}]` : ''}</div>`).join('');
};

/**
 * Mirrors the real `recover()`'s in-flight guard: a Filing a pass is already
 * working on is skipped, so a second concurrent call can never double-create.
 * Without it, React StrictMode's dev double-mount would eat two scripted passes
 * at startup and no [Check again] could ever be observed.
 */
const inFlight = new Set<string>();

async function* recover(only?: string): AsyncGenerator<RecoveryEvent> {
  record(`recover(${only ?? 'all'}) called`);
  for (const plan of plans) {
    if (only !== undefined && plan.id !== only) continue;
    if (inFlight.has(plan.id)) {
      record(`  ↳ ${plan.id} skipped — a look is already in flight`);
      continue;
    }
    inFlight.add(plan.id);
    const n = passCount.get(plan.id) ?? 0;
    passCount.set(plan.id, n + 1);
    const pass = plan.passes[Math.min(n, plan.passes.length - 1)];
    for (const [i, event] of pass.entries()) {
      if (i > 0) await wait(delay);
      yield event;
    }
    // A pass whose script stops at `checking` is a lookup that never answers.
    if (pass[pass.length - 1]?.state === 'checking') await new Promise<void>(() => {});
    inFlight.delete(plan.id);
  }
}
